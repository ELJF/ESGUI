# 递归收集所有子目录
file(GLOB_RECURSE ALL_SUBDIRS LIST_DIRECTORIES true RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} "*")

set(INCLUDE_DIRS "")
foreach(item ${ALL_SUBDIRS})
    if(IS_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/${item})
        list(APPEND INCLUDE_DIRS ${CMAKE_CURRENT_SOURCE_DIR}/${item})
    endif()
endforeach()

target_include_directories(${CMAKE_PROJECT_NAME} PUBLIC ${INCLUDE_DIRS})

file(GLOB_RECURSE SRC_FILES CONFIGURE_DEPENDS "./*.*")
target_sources(${CMAKE_PROJECT_NAME} PRIVATE ${SRC_FILES})
